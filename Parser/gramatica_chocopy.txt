# Gramatica de ChocoPy para el generador LR(1) (formato de lr1/generador).
# Sigue la Figura 3 del Language Reference (sec. 4) y la tabla de precedencia 4.1.
# Los no terminales van en MAYUSCULAS; los terminales van entre llaves y usan
# el mismo nombre que devuelve tokenTypeToString() del scanner, por ejemplo
# {KW_IF} o {IDENTIFIER}. e es la cadena vacia. Las alternativas se separan con
# espacios, asi que los simbolos de una misma alternativa van pegados.
# Las palabras reservadas de Python que ChocoPy no usa (KW_RESERVED) no
# aparecen: el parser las rechaza como error sintactico.

# --- Programa: [var_def | func_def | class_def]* stmt* ---
# Las dos listas son no vacias para que la gramatica sea LR(1) sin conflictos
# (var_def y una sentencia pueden empezar ambos con un IDENTIFIER).
PROGRAMA > e DEFINICIONES SENTLISTA DEFINICIONESSENTLISTA
DEFINICIONES > UNADEF DEFINICIONESUNADEF
UNADEF > VARDEF FUNCDEF CLASEDEF
SENTLISTA > SENTENCIA SENTLISTASENTENCIA

# --- Definiciones ---
VARDEF > VARTIPADA{EQUAL}LITERAL{NEWLINE}
VARTIPADA > {IDENTIFIER}{COLON}TIPO
TIPO > {IDENTIFIER} {IDSTRING} {LBRACKET}TIPO{RBRACKET}

FUNCDEF > {KW_DEF}{IDENTIFIER}{LPAREN}PARAMETROS{RPAREN}RETTIPO{COLON}{NEWLINE}{INDENT}CUERPOFUNC{DEDENT}
PARAMETROS > e LISTAPARAM
LISTAPARAM > VARTIPADA LISTAPARAM{COMMA}VARTIPADA
RETTIPO > e {ARROW}TIPO

# func_body ::= [global_decl | nonlocal_decl | var_def | func_def]* stmt+
CUERPOFUNC > SENTLISTA DECLARACIONESSENTLISTA
DECLARACIONES > UNADECL DECLARACIONESUNADECL
UNADECL > GLOBALDECL NONLOCALDECL VARDEF FUNCDEF
GLOBALDECL > {KW_GLOBAL}{IDENTIFIER}{NEWLINE}
NONLOCALDECL > {KW_NONLOCAL}{IDENTIFIER}{NEWLINE}

CLASEDEF > {KW_CLASS}{IDENTIFIER}{LPAREN}{IDENTIFIER}{RPAREN}{COLON}{NEWLINE}{INDENT}CUERPOCLASE{DEDENT}
CUERPOCLASE > {KW_PASS}{NEWLINE} LISTADECL
LISTADECL > DECLCLASE LISTADECLDECLCLASE
DECLCLASE > VARDEF FUNCDEF

# --- Sentencias ---
SENTENCIA > SIMPLE{NEWLINE} IFSTMT WHILESTMT FORSTMT
SIMPLE > {KW_PASS} {KW_RETURN} {KW_RETURN}EXPR EXPR ASIGNACION
# [target =]+ expr
ASIGNACION > OBJETIVO{EQUAL}EXPR OBJETIVO{EQUAL}ASIGNACION
OBJETIVO > {IDENTIFIER} MIEMBRO INDICE

IFSTMT > {KW_IF}EXPR{COLON}BLOQUELISTAELIFOPCIONSINO
LISTAELIF > e LISTAELIF{KW_ELIF}EXPR{COLON}BLOQUE
OPCIONSINO > e {KW_ELSE}{COLON}BLOQUE
WHILESTMT > {KW_WHILE}EXPR{COLON}BLOQUE
FORSTMT > {KW_FOR}{IDENTIFIER}{KW_IN}EXPR{COLON}BLOQUE

BLOQUE > {NEWLINE}{INDENT}SENTLISTA{DEDENT}

# --- Expresiones (de menor a mayor precedencia, tabla 4.1) ---
EXPR > ORE ORE{KW_IF}ORE{KW_ELSE}EXPR
ORE > ANDE ORE{KW_OR}ANDE
ANDE > NOTE ANDE{KW_AND}NOTE
NOTE > CMPE {KW_NOT}NOTE
CMPE > ARIT ARITCMPOPARIT
CMPOP > {EQ_EQ} {BANG_EQ} {LESS} {GREATER} {LESS_EQ} {GREATER_EQ} {KW_IS}
ARIT > TERM ARIT{PLUS}TERM ARIT{MINUS}TERM
TERM > UNARIO TERM{STAR}UNARIO TERM{SLASH_SLASH}UNARIO TERM{PERCENT}UNARIO
UNARIO > POSTF {MINUS}UNARIO

# member_expr, index_expr y llamadas (solo ID(...) o member_expr(...))
POSTF > PRIM MIEMBRO INDICE LLAMADA
MIEMBRO > POSTF{DOT}{IDENTIFIER}
INDICE > POSTF{LBRACKET}EXPR{RBRACKET}
LLAMADA > {IDENTIFIER}{LPAREN}ARGUMENTOS{RPAREN} MIEMBRO{LPAREN}ARGUMENTOS{RPAREN}
PRIM > {IDENTIFIER} LITERAL {LPAREN}EXPR{RPAREN} {LBRACKET}ELEMENTOS{RBRACKET}
LITERAL > {KW_NONE} {KW_TRUE} {KW_FALSE} {INTEGER} {STRING} {IDSTRING}
ARGUMENTOS > e ARGLISTA
ARGLISTA > EXPR ARGLISTA{COMMA}EXPR
ELEMENTOS > e ELEMLISTA
ELEMLISTA > EXPR ELEMLISTA{COMMA}EXPR
